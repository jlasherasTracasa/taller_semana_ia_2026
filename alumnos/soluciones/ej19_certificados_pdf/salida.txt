import csv
from pypdf import PdfReader

with open("nombres.csv", newline="", encoding="utf-8") as f:
    filas = list(csv.DictReader(f))

def slug(n):
    import re, unicodedata
    t = unicodedata.normalize("NFKD", n)
    t = "".join(c for c in t if not unicodedata.combining(c))
    return re.sub(r"[^A-Za-z0-9]+", "_", t).strip("_").lower()

ok = True
vistos = set()
for fila in filas:
    pdf = f"certificado_asistencia_{slug(fila['nombre'])}.pdf"
    texto = PdfReader(pdf).pages[0].extract_text()
    bien = fila["nombre"] in texto and fila["curso"] in texto and fila["horas"] in texto
    ok &= bien and pdf not in vistos
    vistos.add(pdf)
    print(("OK " if bien else "FALLO ") + pdf)

print("Resultado:", "TODOS CORRECTOS" if ok else "ERROR")
