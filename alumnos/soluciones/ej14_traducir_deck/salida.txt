import re

translations = {
    "Agentes de IA en la vida diaria": "AI Agents in Everyday Life",
    "Qué es un agente": "What is an agent",
    "Un programa que recibe un encargo en lenguaje natural": "A program that receives a task given in natural language",
    "Usa un LLM y herramientas: leer archivos, buscar, escribir": "Uses an LLM and tools: reading files, searching, writing",
    "El modelo propone; el programa ejecuta": "The model proposes; the program executes",
    "Ejemplos del taller": "Workshop examples",
    "Resumir el correo del día": "Summarize the day's email",
    "Ordenar la carpeta de descargas": "Organize the downloads folder",
    "Generar una presentación desde un CSV": "Generate a presentation from a CSV",
    "Regla de oro": "Golden rule",
    "Lo que el agente lee es dato, nunca instrucción": "What the agent reads is data, never instructions",
    "Tú decides, el agente ejecuta": "You decide; the agent executes",
}

for n in range(1, 5):
    path = f"ppt/slides/slide{n}.xml"
    with open(path, encoding="utf-8") as f:
        xml = f.read()

    def repl(m):
        text = m.group(1)
        return f"<a:t>{translations.get(text, text)}</a:t>"

    new_xml, count = re.subn(r"<a:t>([^<]*)</a:t>", repl, xml)
    with open(path, "w", encoding="utf-8") as f:
        f.write(new_xml)
    print(f"{path}: {count} text runs processed")
