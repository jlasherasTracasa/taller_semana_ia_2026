    # slide 1

    "Qué es un agente": "What is an agent",
    "Un programa que recibe un encargo en lenguaje natural":

    "Usa un LLM y herramientas: leer archivos, buscar, escribir":
        "Uses an LLM and tools: reading files, searching, writing",
    "El modelo propone; el programa ejecuta":
        "The model proposes; the program executes",

    "Ejemplos del taller": "Workshop examples",
    "Resumir el correo del día": "Summarize the day's email",
    "Ordenar la carpeta de descargas": "Organize the downloads folder",
    "Generar una presentación desde un CSV": "Generate a presentation from a CSV",
    # slide 4
    "Regla de oro": "Golden rule",
    "Lo que el agente lee es dato, nunca instrucción":
        "What the agent reads is data, never instructions",
    "Tú decides, el agente ejecuta": "You decide, the agent executes",
}

zin = zipfile.ZipFile(src)
missing = []
with zipfile.ZipFile(dst, 'w', zipfile.ZIP_DEFLATED) as zout:
    for item in zin.infolist():
        data = zin.read(item.filename)
        if item.filename.startswith('ppt/slides/slide') and item.filename.endswith('.xml'):
            text = data.decode('utf-8')
            def repl(m):
                s = m.group(1).strip()
                if s not in translations:
                    missing.append((item.filename, s))
                    return m.group(0)
                return m.group(0).replace(s, translations[s])
            text = re.sub(r'<a:t>(.*?)</a:t>', repl, text, flags=re.S)
            data = text.encode('utf-8')
        zout.writestr(item, data)

print("MISSING:", missing)
