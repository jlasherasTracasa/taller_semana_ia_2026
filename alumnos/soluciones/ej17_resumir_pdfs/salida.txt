import pdfplumber, pathlib
out = pathlib.Path("_extract")
out.mkdir(exist_ok=True)
for pdf in sorted(pathlib.Path("apuntes").glob("*.pdf")):
    txt_path = out / (pdf.stem + ".txt")
    with pdfplumber.open(pdf) as doc:
        pages = []
        for i, page in enumerate(doc.pages, 1):
            t = page.extract_text() or ""
            pages.append(f"===== PAGINA {i}/{len(doc.pages)} =====\n{t}")
    txt_path.write_text("\n\n".join(pages), encoding="utf-8")
    print(f"{pdf.name}: {len(doc.pages)} paginas -> {txt_path} ({len(txt_path.read_text())} chars)")
