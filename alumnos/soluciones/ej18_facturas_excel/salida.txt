import re
from datetime import datetime
from pypdf import PdfReader
from openpyxl import Workbook
from openpyxl.styles import Font, Alignment

def parse_pdf(path):
    text = "\n".join(p.extract_text() for p in PdfReader(path).pages)
    emisor = re.search(r"Factura\s+\S+\s+—\s+(.+)", text).group(1).strip()
    fecha = datetime.strptime(re.search(r"Fecha:\s*(\d{2}/\d{2}/\d{4})", text).group(1), "%d/%m/%Y")
    base = float(re.search(r"Base imponible\s*\.?\s*([\d.,]+)\s*EUR", text).group(1).replace(".", "").replace(",", "."))
    iva = float(re.search(r"IVA\s*\d+\s*%\s*\.?\s*([\d.,]+)\s*EUR", text).group(1).replace(".", "").replace(",", "."))
    total = float(re.search(r"TOTAL\s*\.?\s*([\d.,]+)\s*EUR", text).group(1).replace(".", "").replace(",", "."))
    return emisor, fecha, base, iva, total

files = ["facturas/factura_ferreteria.pdf", "facturas/factura_limpieza.pdf"]
rows = [parse_pdf(f) for f in files]

wb = Workbook()
ws = wb.active
ws.title = "Facturas"

headers = ["Emisor", "Fecha", "Base imponible", "IVA", "Total"]
ws.append(headers)
for c in ws[1]:
    c.font = Font(bold=True)
    c.alignment = Alignment(horizontal="center")

money_fmt = '#,##0.00 "EUR"'

El problema está en mis regex: permitía un solo punto (`\.?`) pero las líneas tienen muchos puntos de relleno (`......`). Lo corrijo usando `[\s.]*`:
$ python3 << 'EOF'
import re
from datetime import datetime
from pypdf import PdfReader
from openpyxl import Workbook
from openpyxl.styles import Font, Alignment

def a_num(s):
    return float(s.replace(".", "").replace(",", "."))

def parse_pdf(path):
    text = "\n".join(p.extract_text() for p in PdfReader(path).pages)
    emisor = re.search(r"Factura\s+\S+\s+—\s+(.+)", text).group(1).strip()
    fecha = datetime.strptime(re.search(r"Fecha:\s*(\d{2}/\d{2}/\d{4})", text).group(1), "%d/%m/%Y")
    base = a_num(re.search(r"Base imponible[\s.]*(\d[\d.,]*)\s*EUR", text).group(1))
    iva = a_num(re.search(r"IVA\s+\d+\s*%[\s.]*(\d[\d.,]*)\s*EUR", text).group(1))
    total = a_num(re.search(r"TOTAL[\s.]*(\d[\d.,]*)\s*EUR", text).group(1))
    return emisor, fecha, base, iva, total

files = ["facturas/factura_ferreteria.pdf", "facturas/factura_limpieza.pdf"]
rows = [parse_pdf(f) for f in files]

wb = Workbook()
ws = wb.active
ws.title = "Facturas"

headers = ["Emisor", "Fecha", "Base imponible", "IVA", "Total"]
ws.append(headers)
for c in ws[1]:
    c.font = Font(bold=True)
    c.alignment = Alignment(horizontal="center")

money_fmt = '#,##0.00 "EUR"'
date_fmt = 'DD/MM/YYYY'

for emisor, fecha, base, iva, total in rows:
    ws.append([emisor, fecha, base, iva, total])
    r = ws.max_row
    ws.cell(row=r, column=2).number_format = date_fmt
    for col in (3, 4, 5):
        ws.cell(row=r, column=col).number_format = money_fmt

# Fila final con la suma de totales escrita como número (no como fórmula)
suma_totales = round(sum(r[4] for r in rows), 2)
ws.append(["SUMA TOTAL", None, None, None, suma_totales])
for c in ws[ws.max_row]:
    c.font = Font(bold=True)
ws.cell(row=ws.max_row, column=5).number_format = money_fmt

for col, width in zip("ABCDE", [28, 12, 16, 14, 16]):
    ws.column_dimensions[col].width = width

wb.save("facturas.xlsx")

# Verificación: el total debe ser base + IVA en cada factura,
# y la suma escrita debe coincidir con la suma calculada.
for emisor, fecha, base, iva, total in rows:
    assert abs(base + iva - total) < 0.02, f"Desajuste en {emisor}"
assert abs(suma_totales - sum(r[4] for r in rows)) < 1e-9
print("Verificaciones OK (base+IVA=total en cada factura)")
print(f"Suma de totales (valor numérico escrito): {suma_totales}")
